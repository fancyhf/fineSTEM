=== variants/standard stepModel 全文 ===
function stepModel(previousState, config) {
  const { step, grass, herbivores, water } = previousState;
  const { environment } = config;
  let newGrass = grass;
  if (water > 0) {
    if (grass < 80) { newGrass = grass + 2; } else { newGrass = grass + 0.5; }
  }
  const eaten = herbivores * 0.8;
  newGrass = Math.max(0, newGrass - eaten);
  let newHerbivores = herbivores;
  if (newGrass > 50) { newHerbivores = herbivores + 2; }
  else if (newGrass < 20) { newHerbivores = herbivores - 3; }
  newHerbivores = Math.max(0, Math.min(100, newHerbivores));
  // [干旱岛改造] 差异①：low 从 1 改为 0（无雨档，水只出不进）
  const replenishSpeed = { low: 0, mid: 3, high: 6 }[environment.replenishLevel] ?? 3; // 注意用 ??：0 是假值，|| 会把 0 静默兜底成 3
  let newWater = water + replenishSpeed - (herbivores * 0.3);
  newWater = Math.max(0, Math.min(100, newWater));
  return {
    step: step + 1,
    grass: Math.round(newGrass * 100) / 100,
    herbivores: Math.round(newHerbivores),
    water: Math.round(newWater * 100) / 100
  };
}
